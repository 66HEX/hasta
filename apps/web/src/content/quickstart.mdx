```ts inbox.ts
import { createHastaController } from '@66hex/hasta/vanilla';

const hasta = createHastaController({ horizonMs: 160 });
const target = hasta.target({
  when: { type: 'intersects', withinMs: 160 },
  onPredict: ({ etaMs }) => {
    console.log(`Inbox in ${Math.round(etaMs)} ms`);
  },
});

target.ref(document.querySelector<HTMLButtonElement>('#inbox'));
hasta.connect();

export function dispose() {
  hasta.destroy();
}
```

Run after a button with `id="inbox"` exists. Call `dispose()` when removing the view.

<FrameworkPanel value="react">

```tsx Inbox.tsx
import { useHasta, useHastaTarget } from '@66hex/hasta/react';

export function Inbox() {
  const hasta = useHasta({ horizonMs: 160 });
  const target = useHastaTarget(hasta, {
    when: { type: 'intersects', withinMs: 160 },
    onPredict: ({ etaMs }) => {
      console.log(`Inbox in ${Math.round(etaMs)} ms`);
    },
  });

  return <button ref={target.ref}>Inbox</button>;
}
```
